# GTK3 looks for an actual theme folder named "Adwaita-dark", it does not
# derive one from gtk-application-prefer-dark-theme the way libadwaita (GTK4)
# apps do. Without this package the name in settings.ini/gsettings resolves
# to nothing and GTK3 apps -- Thunar included -- silently fall back to light
# Adwaita no matter what those files say.
gnome-themes-extra
